import java.util.Date;
import java.util.ArrayList;

class Save{
    private final String version;
    private final Date date;
    
    public Save(String version){
        this.version = version;
        this.date = new Date();
    }
    
    public String getVersion(){
        return version;
    }
    
    public Date getDate(){
        return date;
    }
}

class Project{
    private String version;
    private Date date;
    
    public void setVersionAndDate(String version){
        this.version = version;
        this.date = new Date();
    }
    
    public Save save(){
        return new Save(version);
    }
    
    public void load(Save save){
        this.version = save.getVersion();
        this.date = save.getDate();
    }
    
    @Override
    public String toString(){
        return "Project: "+ "\nversion = '"+version+ '\''+"\nDate = "+date+"\n";
    }
}

class GitHubRepo{
    private Save save;
    
    public Save getSave(){
        return save;
    }
    
    public void setSave(Save save){
        this.save = save;
    }
}

public class Main{
	public static void main(String[] args) throws InterruptedException {
        Project project  = new Project();
        GitHubRepo ghr = new GitHubRepo();
        
        System.out.println("Creating project");
        project.setVersionAndDate("Version 1.0");
        ghr.setSave(project.save());
        System.out.println("Saving project");
        System.out.println(project);
        
        
        System.out.println("Updating project to Version 1.1");
        Thread.sleep(5000);
        project.setVersionAndDate("Version 1.1");
        System.out.println(project);
        
        System.out.println("Rolling back project to v1.0");
        project.load(ghr.getSave());
        System.out.println(project);
        
	}
}
