public class Main {
    public static void main(String[] args) {
        Day currentDay = new Monday();
        WeekSchedule schedule = new WeekSchedule();
        schedule.setDay(currentDay);
        
        for (int i = 0; i < 7; i++) {
            schedule.whatToDo();
            schedule.nextDay();
        }
    }
}

interface Day {
    void whatToDo();
}

class Monday implements Day {
    public void whatToDo() { System.out.println("Понедельник: Начать новую неделю"); }
}

class Tuesday implements Day {
    public void whatToDo() { System.out.println("Вторник: Работа над проектом"); }
}

class Wednesday implements Day {
    public void whatToDo() { System.out.println("Среда: Середина недели, встреча с командой"); }
}

class Thursday implements Day {
    public void whatToDo() { System.out.println("Четверг: Завершение задач"); }
}

class Friday implements Day {
    public void whatToDo() { System.out.println("Пятница: Подведение итогов"); }
}

class Saturday implements Day {
    public void whatToDo() { System.out.println("Суббота: Отдых с семьей"); }
}

class Sunday implements Day {
    public void whatToDo() { System.out.println("Воскресенье: Подготовка к новой неделе"); }
}

class WeekSchedule {
    private Day currentDay;
    
    public void setDay(Day day) {
        this.currentDay = day;
    }
    
    public void nextDay() {
        if (currentDay instanceof Monday) {
            setDay(new Tuesday());
        } else if (currentDay instanceof Tuesday) {
            setDay(new Wednesday());
        } else if (currentDay instanceof Wednesday) {
            setDay(new Thursday());
        } else if (currentDay instanceof Thursday) {
            setDay(new Friday());
        } else if (currentDay instanceof Friday) {
            setDay(new Saturday());
        } else if (currentDay instanceof Saturday) {
            setDay(new Sunday());
        } else if (currentDay instanceof Sunday) {
            setDay(new Monday());
        }
    }
    
    public void whatToDo() {
        currentDay.whatToDo();
    }
}
