public class Main {
    public static void main(String[] args) {
        Vehicle vehicle = new Vehicle();
        
        vehicle.setMode(new Walking());
        vehicle.executeMode();
        
        vehicle.setMode(new Biking());
        vehicle.executeMode();
        
        vehicle.setMode(new Driving());
        vehicle.executeMode();
        
        vehicle.setMode(new Flying());
        vehicle.executeMode();
    }
}

interface TransportMode {
    void move();
}

class Walking implements TransportMode {
    public void move() { System.out.println("Иду пешком (скорость 5 км/ч)"); }
}

class Biking implements TransportMode {
    public void move() { System.out.println("Еду на велосипеде (скорость 20 км/ч)"); }
}

class Driving implements TransportMode {
    public void move() { System.out.println("Еду на машине (скорость 60 км/ч)"); }
}

class Flying implements TransportMode {
    public void move() { System.out.println("Лечу на самолете (скорость 900 км/ч)"); }
}

class Vehicle {
    private TransportMode mode;
    
    public void setMode(TransportMode mode) {
        this.mode = mode;
    }
    
    public void executeMode() {
        mode.move();
    }
}
