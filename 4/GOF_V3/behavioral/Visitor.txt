interface Architect {
    void design(Foundation foundation);
    void design(Walls walls);
    void design(Roof roof);
}

interface BuildingElement {
    void beDesigned(Architect architect);
}

class Foundation implements BuildingElement {
    @Override
    public void beDesigned(Architect architect) {
        architect.design(this);
    }
}

class Walls implements BuildingElement {
    @Override
    public void beDesigned(Architect architect) {
        architect.design(this);
    }
}

class Roof implements BuildingElement {
    @Override
    public void beDesigned(Architect architect) {
        architect.design(this);
    }
}

class JuniorArchitect implements Architect {
    @Override
    public void design(Foundation f) {
        System.out.println("Фундамент: простой ленточный (проект Junior)");
    }
    
    @Override
    public void design(Walls w) {
        System.out.println("Стены: кирпичные, стандартные (проект Junior)");
    }
    
    @Override
    public void design(Roof r) {
        System.out.println("Крыша: двускатная, шифер (проект Junior)");
    }
}

class SeniorArchitect implements Architect {
    @Override
    public void design(Foundation f) {
        System.out.println("Фундамент: монолитная плита с усилением (проект Senior)");
    }
    
    @Override
    public void design(Walls w) {
        System.out.println("Стены: газобетон с утеплением (проект Senior)");
    }
    
    @Override
    public void design(Roof r) {
        System.out.println("Крыша: мансардная, металлочерепица (проект Senior)");
    }
}

class BuildingProject implements BuildingElement {
    private final BuildingElement[] elements;
    
    public BuildingProject() {
        elements = new BuildingElement[] {
            new Foundation(), new Walls(), new Roof()
        };
    }
    
    @Override
    public void beDesigned(Architect architect) {
        for (BuildingElement element : elements) {
            element.beDesigned(architect);
        }
    }
}

public class Main {
    public static void main(String[] args) {
        BuildingProject project = new BuildingProject();
        Architect junior = new JuniorArchitect();
        Architect senior = new SeniorArchitect();
        
        System.out.println("Проект от Junior архитектора:");
        project.beDesigned(junior);
        
        System.out.println("\nПроект от Senior архитектора:");
        project.beDesigned(senior);
    }
}
